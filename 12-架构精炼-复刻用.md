## 二、数据库设计（8 张核心表）

### 完整 Prisma Schema

```prisma
// ===== 三级记忆架构 =====

model Organization {
  id                  String   @id @default(uuid())
  name                String
  legalPerson         String?  @map("legal_person")
  businessLicense     String?  @map("business_license")
  icpFilingNo         String?  @map("icp_filing_no")
  complianceStrategy  Json     @default("{}") @map("compliance_strategy")
  createdAt           DateTime @default(now()) @map("created_at")
  updatedAt           DateTime @updatedAt @map("updated_at")
  workspaces          Workspace[]
  @@map("organizations")
}

model Workspace {
  id             String   @id @default(uuid())
  organizationId String   @map("organization_id")
  productName    String   @map("product_name")
  productType    String   @map("product_type") // app|miniprogram|web|api
  industry       String   // ai_marketing|ai_content|ai_companion|...
  riskProfile    Json     @default("{}") @map("risk_profile")
  filingStatus   Json     @default("{}") @map("filing_status")
  createdAt      DateTime @default(now()) @map("created_at")
  updatedAt      DateTime @updatedAt @map("updated_at")
  organization   Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)
  sessions       Session[]
  operationLogs  OperationLog[]
  calendarEvents CalendarEvent[]
  @@index([organizationId])
  @@map("workspaces")
}

model Session {
  id             String   @id @default(uuid())
  workspaceId    String   @map("workspace_id")
  inputSnapshot  Json     @map("input_snapshot")
  outputType     String   @map("output_type") // risk_brief|review_report|doc_package|mock_filing
  outputData     Json?    @map("output_data")
  triggeredRules Json     @default("[]") @map("triggered_rules")
  createdAt      DateTime @default(now()) @map("created_at")
  workspace      Workspace @relation(fields: [workspaceId], references: [id], onDelete: Cascade)
  @@index([workspaceId])
  @@index([createdAt(sort: Desc)])
  @@map("sessions")
}

model OperationLog {
  id          String   @id @default(uuid())
  userId      String   @map("user_id")
  workspaceId String   @map("workspace_id")
  sessionId   String?  @map("session_id")
  actionType  String   @map("action_type") // assess|review|generate|filing|view
  ruleIds     Json     @default("[]") @map("rule_ids")
  legalRefs   Json     @default("[]") @map("legal_refs")
  metadata    Json     @default("{}")
  createdAt   DateTime @default(now()) @map("created_at")
  workspace   Workspace @relation(fields: [workspaceId], references: [id], onDelete: Cascade)
  @@index([userId])
  @@index([workspaceId])
  @@map("operation_logs")
}

model CalendarEvent {
  id          String   @id @default(uuid())
  workspaceId String   @map("workspace_id")
  title       String
  eventType   String   @map("event_type") // filing_deadline|regulatory_update|todo|milestone
  dueDate     DateTime @map("due_date")
  status      String   @default("pending") // pending|completed|overdue
  workspace   Workspace @relation(fields: [workspaceId], references: [id], onDelete: Cascade)
  @@map("calendar_events")
}

// ===== 法规数据库 =====

model Regulatory {
  id              String   @id @default(uuid())
  title           String
  shortTitle      String?  @map("short_title")
  issuingBody     String   @map("issuing_body")
  lawType         String   @map("law_type") // law|administrative_regulation|departmental_rule
  legalLevel      String   @map("legal_level")
  publishDate     DateTime @map("publish_date")
  effectiveDate   DateTime @map("effective_date")
  status          String   @default("effective")
  category        String   @default("general") // ai|data_privacy|cybersecurity|consumer|advertising
  keywords        String[] @default([])
  fullText        String?
  penaltyRefs     Json     @default("[]")
  version         Int      @default(1)
  articles        RegulatoryArticle[]
  @@map("regulations")
}

model RegulatoryArticle {
  id            String   @id @default(uuid())
  regulationId  String   @map("regulation_id")
  articleOrder  Int      @map("article_order")
  articleNumber String   @map("article_number") // "第4条"
  content       String                        // 条款全文
  obligations   Json     @default("[]")        // [{type, target, triggerKeywords, severity}]
  penalty       Json?                          // {type, description, statutoryPenalty}
  keywords      String[] @default([])
  regulation    Regulatory @relation(fields: [regulationId], references: [id], onDelete: Cascade)
  @@map("regulatory_articles")
}

// ===== 合规分析记录 =====

model ComplianceAnalysis {
  id              String   @id @default(uuid())
  workspaceId     String   @map("workspace_id")
  sessionId       String?  @map("session_id")
  analysisType    String   @map("analysis_type") // risk_assessment|compliance_review|deep_analysis
  inputData       Json     @map("input_data")
  matchedArticles Json     @default("[]") @map("matched_articles")
  riskScore       Float    @default(0) @map("risk_score")
  riskLevel       String   @default("low") @map("risk_level")
  findings        Json     @default("[]")
  summary         String?
  workspace       Workspace @relation(fields: [workspaceId], references: [id], onDelete: Cascade)
  @@map("compliance_analyses")
}
```

## 三、合规引擎架构

### 引擎栈（三层并存，需统一接入）

```
旧引擎 (rules.ts)         → /api/assess/risk
  17 条硬编码规则           /api/review/compliance
  简单关键词匹配            前端已接入 ✅

新引擎 (compliance-engine.ts) → /api/compliance/analyze
  9 部法规 × 22 条条款       四层管道：提取→匹配→评分→报告
  0-100 风险评分              前端 /demo 页部分接入 ⚠️

Agent 引擎 (agents/)      → /api/compliance/agent-analyze
  Decomposer → Matcher     需求拆解 → 法规定位 → 差距分析
  → GapAnalyzer → Report   知识图谱联动
                            前端 /demo 页部分接入 ⚠️
```

### 四层分析管道（新引擎核心）

```
INPUT LAYER ─────────────────────────────────────────
  extractFeatures(productName, description, formFields)
  → ComplianceFeatures (38 个特征维度)
  → 识别 AI 技术路径、产品形态、目标用户、数据收集、内容类型等

PARSING LAYER ───────────────────────────────────────
  1. 产品描述分词 + 特征标签化
  2. AI 技术路径识别：llm_api | fine_tuned | rag | deep_synthesis
  3. 场景归类 → 法域确定（中国 → 仅匹配中国法规）

MATCHING LAYER ───────────────────────────────────────
  Level 1 — 关键词精确匹配：triggerKeywords 包含式匹配
  Level 2 — 语义特征匹配：特征标签 → 义务类型映射
    收集个人信息 → 触发隐私政策告知义务（个保法第17条）
    敏感信息 → 触发单独同意义务（个保法第28条）
    未成年人 → 触发未成年人保护（未保法第75条）
  Level 3 — 跨条款关联推导：
    生成式AI → 自动关联内容标识义务（第12条）
    深度合成 → 自动关联深度合成标识（第16条）
    AI + 营销 → 关联消费者保护（消保法第20条）

JUDGMENT LAYER ───────────────────────────────────────
  风险评分 = Σ(义务权重 × 严重乘数 + 处罚附加分)

  obligationWeights: mandatory=25, prohibitive=20, conditional=15, procedural=10
  severityMultipliers: critical=2.0, high=1.5, medium=1.0, low=0.5
  penaltyBonus: 刑事+20, 吊销许可+15, 停业+10, 罚款+5

  风险等级: 0-15=low, 16-30=medium, 31-50=high, 51-100=critical

  结论判定:
    blocked:    critical>=2 OR riskScore>=51
    conditional: high>=2 OR 31<=riskScore<51 OR medium>=3
    approved:   其余

OUTPUT LAYER ────────────────────────────────────────
  ComplianceReport 包含：
  - 评审结论 + 风险评分 (0-100)
  - 四维风险分解（政策/行业/产品/数据）
  - 备案判断清单（含急迫度）
  - 安全评估判断（含理由+法规依据）
  - 整改清单（P0/P1 优先级 + 截止时间）
  - 低风险试点版本建议
  - 处罚风险提示（含最高处罚金额）
  - 数据溯源（法规库版本 + 分析时间 + 触发条款数）
```

### Demo 全部 API

| 方法 | 路径 | 功能 | 引擎 |

| `GET` | `/api/regulatory/search` | 法规检索 | 法规库 |

```typescript
interface ComplianceRule {
```
