```typescript
// lib/risk-calculator.ts
// Demo 简化风险计算 —— 纯前端，不含核心引擎逻辑

import type { RiskLevel } from '@/types/domain';

export type Answer = 'yes' | 'no' | 'na';

export type Question = {
  id: number;
  text: string;
};

export type RiskDimension = {
  title: string;
  level: RiskLevel;
  detail: string;
};

export type RiskResult = {
  riskLevel: RiskLevel;
  summary: string;
  dimensions: RiskDimension[];
  suggestions: string[];
  materials: string[];
};

export const QUESTIONS: Question[] = [
  { id: 1, text: '产品主要功能是什么？' },
  { id: 2, text: '是否面向中国大陆用户？' },
  { id: 3, text: '是否生成文本、图片、音频或视频？' },
  { id: 4, text: '是否存在推荐、排序或个性化分发？' },
  { id: 5, text: '是否涉及深度合成（换脸、语音合成等）？' },
  { id: 6, text: '是否收集个人信息？' },
  { id: 7, text: '是否处理敏感个人信息？' },
  { id: 8, text: '是否可能触达未成年人？' },
  { id: 9, text: '是否涉及广告、金融、医疗、教育、陪伴等高敏行业？' },
  { id: 10, text: '上线前是否已有 ICP 或主体备案信息？' },
];

const LEVEL_SCORE: Record<RiskLevel, number> = { '低': 0, '中': 1, '高': 2, '谨慎': 3, '禁止': 4 };

function maxLevel(a: RiskLevel, b: RiskLevel): RiskLevel {
  return LEVEL_SCORE[a] >= LEVEL_SCORE[b] ? a : b;
}

export function calcRisk(answers: Record<number, Answer>): RiskResult {
  const dims: Record<string, { level: RiskLevel; detail: string }> = {
    policy:   { level: '低', detail: '政策风险：当前无明显政策风险。' },
    industry: { level: '低', detail: '行业风险：所在行业无明显特殊监管。' },
    product:  { level: '低', detail: '产品形态未触及高风险 AI 领域。' },
    data:     { level: '低', detail: '无可识别的数据合规高风险信号。' },
  };

  if (answers[2] === 'yes') {
    dims.policy = { level: '中', detail: '面向中国大陆用户，须遵守中国 AI 监管体系（《生成式AI暂行办法》《算法推荐管理规定》等）。' };
  }
  if (answers[3] === 'yes') {
    dims.product = { level: '中', detail: '涉及 AI 内容生成，可能触发《生成式人工智能服务管理暂行办法》。' };
  }
  if (answers[4] === 'yes') {
    dims.product.level = maxLevel(dims.product.level, '高');
    dims.product.detail = '涉及算法推荐/个性化分发，触发《算法推荐管理规定》，需算法备案。';
  }
  if (answers[5] === 'yes') {
    dims.product.level = '高';
    dims.product.detail = '涉及深度合成，须遵守《深度合成管理规定》第16、19条。';
    dims.policy.level = maxLevel(dims.policy.level, '高');
  }
  if (answers[6] === 'yes') {
    dims.data = { level: '中', detail: '收集个人信息，触发《个人信息保护法》第13、17条告知同意义务。' };
  }
  if (answers[7] === 'yes') {
    dims.data = { level: '高', detail: '涉及敏感个人信息，触发《个人信息保护法》第28条单独同意义务。' };
  }
  if (answers[8] === 'yes') {
    dims.industry = { level: '高', detail: '可能触达未成年人，触发《未成年人保护法》第75条。' };
  }
  if (answers[9] === 'yes') {
    dims.industry.level = maxLevel(dims.industry.level, '中');
    dims.industry.detail = '涉及高敏行业，可能叠加行业监管要求。';
  }
  if (answers[10] === 'no') {
    dims.policy.level = maxLevel(dims.policy.level, '中');
    dims.policy.detail += ' 尚无 ICP 备案，需在上线前完成。';
  }

  const maxDimScore = Math.max(
    LEVEL_SCORE[dims.policy.level],
    LEVEL_SCORE[dims.industry.level],
    LEVEL_SCORE[dims.product.level],
    LEVEL_SCORE[dims.data.level],
  );

  let overallLevel: RiskLevel;
  if (maxDimScore <= 0) overallLevel = '低';
  else if (maxDimScore === 1) overallLevel = '中';
  else if (maxDimScore === 2) overallLevel = '高';
  else overallLevel = '谨慎';

  const suggestions: string[] = [];
  const materials: string[] = [];

  if (answers[3] === 'yes' || answers[4] === 'yes' || answers[5] === 'yes') {
    suggestions.push('明确 AI 服务属性，在显著位置标注「AI 生成」');
    materials.push('AI 服务特别说明');
  }
  if (answers[6] === 'yes') {
    materials.push('隐私政策', '用户协议');
  }
  if (answers[7] === 'yes') {
    suggestions.push('为敏感个人信息处理提供单独同意机制');
    materials.push('单独同意弹窗文案');
  }
  if (answers[8] === 'yes') {
    suggestions.push('建立年龄验证机制，限制未成年人使用范围');
    materials.push('未成年人保护声明');
  }
  if (answers[4] === 'yes' || answers[5] === 'yes') {
    suggestions.push('启动算法备案 / 深度合成备案流程');
    materials.push('算法自评估报告');
  }
  if (answers[10] === 'no') {
    suggestions.push('尽快完成 ICP 备案和主体备案');
    materials.push('ICP 备案材料');
  }

  if (suggestions.length === 0) suggestions.push('当前风险较低，建议定期关注法规更新。');
  if (materials.length === 0) materials.push('暂无必须材料');

  const summaries: Record<RiskLevel, string> = {
    '低': '该产品经初步评估风险较低，可在完成基础合规准备后上线。',
    '中': '该产品存在中等合规风险，建议完成备案和文档准备后上线。',
    '高': '该产品存在较高合规风险，需完成备案、安全评估并制定整改方案后方可上线。',
    '谨慎': '该产品处于谨慎区间，建议降低功能范围或在法务指导下分阶段上线。',
    '禁止': '该产品方向可能存在禁止性风险，建议重新评估产品方案。',
  };

  return {
    riskLevel: overallLevel,
    summary: summaries[overallLevel],
    dimensions: [
      { title: '政策风险', ...dims.policy },
      { title: '行业风险', ...dims.industry },
      { title: '产品合规', ...dims.product },
      { title: '数据风险', ...dims.data },
    ],
    suggestions,
    materials,
  };
}
```

#### 1.2 新建文件 `app/part-a/PartAClient.tsx`

这是交互式答题组件的完整代码：

```tsx
'use client';

import { useState, useMemo } from 'react';
import { AppShell } from '@/components/app-shell';
import { Card, CitationBlock, PageTitle, RiskBadge } from '@/components/ui';
import { getProductById } from '@/lib/products';
import { QUESTIONS, calcRisk, type Answer } from '@/lib/risk-calculator';
import { citations } from '@/lib/demo-data';

type Props = { product?: string };

const DEFAULT_ANSWERS: Record<number, Answer> = {
  1: 'yes', 2: 'yes', 3: 'yes', 4: 'yes', 5: 'no',
  6: 'yes', 7: 'no', 8: 'no', 9: 'no', 10: 'no',
};

export default function PartAClient({ product }: Props) {
  const pid = product ?? 'geo-copywriter';
  const productData = getProductById(pid);

  const [answers, setAnswers] = useState<Record<number, Answer>>(() => DEFAULT_ANSWERS);
  const result = useMemo(() => calcRisk(answers), [answers]);

  const handleChange = (qId: number, value: Answer) => {
    setAnswers((prev) => ({ ...prev, [qId]: value }));
  };

  return (
    <AppShell productId={productData.id}>
      <PageTitle
        eyebrow="PART A"
        title="设计期风险初筛"
        description="回答 10 个引导问题，即时生成《AI 产品立项前风险简报》。点击选项切换答案，风险结果实时更新。"
      />

      <div className="grid gap-6 lg:grid-cols-[0.75fr_1.25fr]">
        <Card>
          <h2 className="text-xl font-black">10 题引导</h2>
          <p className="mt-1 text-xs text-muted">点击选项切换答案，结果实时联动更新</p>
          <div className="mt-4 space-y-2">
            {QUESTIONS.map((q) => {
              const current = answers[q.id] ?? 'na';
              return (
                <div key={q.id} className="rounded-2xl border border-sandDeep bg-cream p-3">
                  <div className="text-sm font-semibold">
                    <span className="text-brand">{q.id}.</span> {q.text}
                  </div>
                  <div className="mt-2 flex gap-2">
                    {(['yes', 'no'] as Answer[]).map((opt) => (
                      <button
                        key={opt}
                        onClick={() => handleChange(q.id, opt)}
                        className={`rounded-full px-4 py-1.5 text-xs font-bold transition ${
                          current === opt
                            ? 'bg-brand text-white shadow-warm'
                            : 'bg-white text-muted hover:bg-brandSoft'
                        }`}
                      >
                        {opt === 'yes' ? '是' : '否'}
                      </button>
                    ))}
                  </div>
                </div>
              );
            })}
          </div>
        </Card>

        <Card>
          <div className="flex flex-wrap items-center justify-between gap-3">
            <div>
              <div className="text-sm font-bold text-brand">《AI 产品立项前风险简报》</div>
              <h2 className="mt-1 text-2xl font-black">{productData.name}</h2>
            </div>
            <RiskBadge level={result.riskLevel} />
          </div>
          <p className="mt-5 rounded-2xl bg-brandSoft p-4 text-sm font-semibold leading-7 text-brandDark">
            {result.summary}
          </p>
          <div className="mt-6 grid gap-4 md:grid-cols-2">
            {result.dimensions.map((dim) => (
              <div key={dim.title} className="rounded-3xl border border-sandDeep p-4">
                <div className="flex items-center justify-between">
                  <h3 className="font-black">{dim.title}</h3>
                  <RiskBadge level={dim.level} />
                </div>
                <p className="mt-3 text-sm leading-6 text-muted">{dim.detail}</p>
                <CitationBlock citation={citations.genAI} />
              </div>
            ))}
          </div>
        </Card>
      </div>

      <div className="mt-6 grid gap-6 lg:grid-cols-2">
        <Card>
          <h3 className="text-xl font-black">设计调整建议</h3>
          <ul className="mt-4 list-disc space-y-2 pl-5 text-sm text-muted">
            {result.suggestions.map((s) => <li key={s}>{s}</li>)}
          </ul>
        </Card>
        <Card>
          <h3 className="text-xl font-black">上线前材料清单</h3>
          <ul className="mt-4 list-disc space-y-2 pl-5 text-sm text-muted">
            {result.materials.map((m) => <li key={m}>{m}</li>)}
          </ul>
        </Card>
      </div>
    </AppShell>
  );
}
```

### 任务 2：Part C 复制草案按钮生效

**目标：** 当前「复制草案」按钮无 onClick 处理函数，点击无效。需要接入 `navigator.clipboard.writeText()` 实现复制功能。

#### 2.1 新建文件 `app/part-c/PartCClient.tsx`

```tsx
'use client';

import { useState } from 'react';
import { AppShell } from '@/components/app-shell';
import { Card, PageTitle } from '@/components/ui';
import { getProductById } from '@/lib/products';

type Props = { product?: string };

export default function PartCClient({ product }: Props) {
  const productData = getProductById(product);
  const [copiedIdx, setCopiedIdx] = useState<number | null>(null);

  const handleCopy = async (docType: string, idx: number) => {
    const text = [
      `【${docType}】—— Demo Mock 草案`,
      `本文件基于「${productData.name}」的产品画像、风险简报和开发评审单自动生成。`,
      '服务提供者应明确 AI 服务属性、用户权利义务、数据处理规则、禁止用途和必要备案材料。',
      '',
      '⚠ 本文档为 Demo Mock，不构成正式法律意见。',
    ].join('\n');

    try {
      await navigator.clipboard.writeText(text);
      setCopiedIdx(idx);
      setTimeout(() => setCopiedIdx(null), 2000);
    } catch {
      alert('复制失败，请手动复制');
    }
  };

  return (
    <AppShell productId={productData.id}>
      <PageTitle
        eyebrow="PART C"
        title="上线合规文档中心"
        description="读取 PART A 风险画像和 PART B 评审结论，一键生成前台文本和备案 Mock 包，并标注需要企业补充的占位项。"
      />
      <div className="mb-6 rounded-3xl border border-sandDeep bg-white/70 p-5 shadow-soft">
        <div className="text-sm font-bold text-brand">当前产品</div>
        <div className="mt-1 text-2xl font-black">{productData.name}</div>
        <div className="mt-2 text-sm text-muted">文档将根据当前场景的风险画像、备案判断与整改清单生成。</div>
      </div>
      <div className="grid gap-6 md:grid-cols-2">
        {productData.partC.documents.map((doc, idx) => (
          <Card key={doc.type}>
            <div className="text-sm font-bold text-brand">可生成文档</div>
            <h2 className="mt-2 text-2xl font-black">{doc.type}</h2>
            <p className="mt-3 text-sm leading-6 text-muted">{doc.description}</p>
            <div className="mt-5 rounded-2xl border border-dashed border-sandDeep bg-cream p-4 text-sm leading-7 text-muted">
              <b>文档草案片段：</b><br />
              本文件基于「{productData.name}」的产品画像、风险简报和开发评审单自动生成。服务提供者应明确 AI 服务属性、用户权利义务、数据处理规则、禁止用途和必要备案材料。本文档为 Demo Mock，不构成正式法律意见。
            </div>
            <div className="mt-4 flex flex-wrap gap-2">
              {doc.placeholders.map((x) => (
                <span key={x} className="rounded-full bg-brandSoft px-3 py-1 text-xs font-bold text-brandDark ring-1 ring-brand/15">
                  待补充：{x}
                </span>
              ))}
            </div>
            <button
              onClick={() => handleCopy(doc.type, idx)}
              className={`mt-5 rounded-2xl px-4 py-3 text-sm font-black text-white transition ${
                copiedIdx === idx ? 'bg-green-600' : 'bg-ink hover:bg-brand'
              }`}
            >
              {copiedIdx === idx ? '已复制' : '复制草案'}
            </button>
          </Card>
        ))}
      </div>
    </AppShell>
  );
}
```
